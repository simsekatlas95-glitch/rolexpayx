        function secureGateLogin() {
            let sessionFlag = sessionStorage.getItem('rolex_session_verified');
            if(sessionFlag === 'VERIFIED') return;

            let emailInput = prompt("🔑 Rolexpay Güvenlik Kapısı\nGiriş Kimliğinizi (E-Posta) Yazın kanka:");
            if(!emailInput) { secureLogout(); return; }

            let users = JSON.parse(localStorage.getItem('rolex_users_list')) || [];
            let foundUser = users.find(u => u.email === emailInput.trim().toLowerCase());

            if(!foundUser) { alert("❌ Bu isimde bir kullanıcı sistemde kayıtlı değil!"); secureLogout(); return; }
            if(foundUser.status === 'PASİF') { alert("❌ Erişiminiz engellenmiştir!"); secureLogout(); return; }

            let passInput = prompt(`🔑 Şifrenizi giriniz kanka (${foundUser.name}):`);
            if(passInput !== foundUser.password) { alert("❌ Yanlış şifre girdin!"); secureLogout(); return; }

            currentUser = foundUser.email;
            currentUserRole = foundUser.role;
            localStorage.setItem('rolex_current_user', foundUser.email);
            localStorage.setItem('rolex_current_role', foundUser.role);
            sessionStorage.setItem('rolex_session_verified', 'VERIFIED');

            applyRoleSecurity();
        }

        function secureLogout() {
            sessionStorage.removeItem('rolex_session_verified');
            localStorage.removeItem('rolex_current_user');
            localStorage.removeItem('rolex_current_role');
            window.location.reload();
        }

        function createPendingSignal(bankName, holderName, targetAdmin) {
            let userInput = prompt(`💵 ${holderName} (${bankName}) hesabına gönderilen işlem tutarını yaz kanka:`);
            if (userInput === null) return; 
            let amount = parseFloat(userInput.replace(/[^0-9.]/g, ''));
            if (isNaN(amount) || amount <= 0) { alert("Geçerli bir rakam girmedin kanka!"); return; }

            let senderName = prompt("👤 Parayı gönderen müşterinin Ad Soyad bilgisini yaz kanka:");
            if (!senderName) { alert("Müşteri adı zorunludur!"); return; }

            let txid = prompt("📋 Banka dekont / Referans numarasını yaz kanka:");
            if (!txid) { alert("Dekont numarası dolandırıcılık önlemi için zorunludur!"); return; }

            let pendingSignals = JSON.parse(localStorage.getItem('rolex_pending_signals')) || [];
            
            let users = JSON.parse(localStorage.getItem('rolex_users_list')) || [];
            let me = users.find(u => u.email === currentUser);
            let terminalName = me ? me.name : currentUser;

            pendingSignals.push({
                id: Date.now(),
                bank: bankName,
                holder: holderName,
                admin: targetAdmin,
                senderEmail: currentUser, 
                senderName: terminalName,
                clientName: senderName.trim(),
                txid: txid.trim(),
                amount: amount,
                time: new Date().toLocaleString('tr-TR')
            });

            localStorage.setItem('rolex_pending_signals', JSON.stringify(pendingSignals));
            alert("🚀 İşlem Onay Masasına düştü kanka! Admin kontrol edip onaylayınca ciroya geçecek.");
            
            applyRoleSecurity();
        }

        function approveSignal(id) {
            let pendingSignals = JSON.parse(localStorage.getItem('rolex_pending_signals')) || [];
            let signal = pendingSignals.find(s => s.id === id);
            if(!signal) return;

            let ciroData = JSON.parse(localStorage.getItem('rolex_admin_ciros')) || {};
            if (!ciroData[signal.admin]) ciroData[signal.admin] = { ciro: 0, adet: 0 };
            ciroData[signal.admin].ciro += signal.amount;
            ciroData[signal.admin].adet += 1;
            localStorage.setItem('rolex_admin_ciros', JSON.stringify(ciroData));

            let currentTimestamp = new Date().toLocaleString('tr-TR');

            let transferHistory = JSON.parse(localStorage.getItem('rolex_transfer_history')) || [];
            transferHistory.push({
                bank: signal.bank,
                holder: signal.holder,
                admin: signal.admin,
                senderEmail: signal.senderEmail,
                senderName: signal.senderName,
                amount: signal.amount,
                time: currentTimestamp + " (ONAYLANDI)"
            });
            localStorage.setItem('rolex_transfer_history', JSON.stringify(transferHistory));

            pendingSignals = pendingSignals.filter(s => s.id !== id);
            localStorage.setItem('rolex_pending_signals', JSON.stringify(pendingSignals));

            let sKomisyon = signal.amount * 0.02;
            let formattedAmount = signal.amount.toLocaleString('tr-TR') + " ₺";

            const token = localStorage.getItem('rolex_bot_token') || '';
            const chatId = localStorage.getItem('rolex_chat_id') || '';
            if(token && chatId) {
                const message = `💰 *ROLEXPAY - %100 ONAYLI NAKİT GİRİŞİ* 💰\n\n👤 *Onaylayan Kasa:* ${signal.admin}\n🎯 *Atıcı Terminali:* ${signal.senderName}\n🏛 *Banka:* ${signal.bank}\n👤 *Hesap:* ${signal.holder}\n👤 *Müşteri:* ${signal.clientName}\n📋 *TXID / Dekont:* \`${signal.txid}\`\n💵 *Onaylanan Net Tutar:* ₺${signal.amount.toLocaleString('tr-TR')}\n⚡ *Rolexpay Payı (%2):* ₺${sKomisyon.toLocaleString('tr-TR', {minimumFractionDigits: 2})}\n\n📈 *Toplam Kasa Net Cirosu:* ₺${ciroData[signal.admin].ciro.toLocaleString('tr-TR')}`;
                fetch(`https://api.telegram.org/bot${token}/sendMessage`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ chat_id: chatId, text: message, parse_mode: 'Markdown' }) }).catch(e => {});
            }

            const ilanToken = localStorage.getItem('rolex_ilan_bot_token') || '';
            const ilanChatId = localStorage.getItem('rolex_ilan_chat_id') || '';
            if(ilanToken && ilanChatId) {
                const ilanMessage = `📣 *ROLEXPAY - TRANSFER BİLDİRİMİ* 📣\n\n🎯 *Atıcı Terminali:* \`${signal.senderEmail}\`\n👑 *Kasa Sahibi Admin:* \`${signal.admin}\`\n📅 *Tarih / Saat:* \`${currentTimestamp}\`\n\nArasında *${formattedAmount}* transfer gerçekleştirilmiştir.`;
                fetch(`https://api.telegram.org/bot${ilanToken}/sendMessage`, { 
                    method: 'POST', 
                    headers: { 'Content-Type': 'application/json' }, 
                    body: JSON.stringify({ chat_id: ilanChatId, text: ilanMessage, parse_mode: 'Markdown' }) 
                }).catch(e => {});
            }

            applyRoleSecurity();
        }

        function formatIban(input) {
            let value = input.value.replace(/\s+/g, '').toUpperCase();
            if(!value.startsWith('TR')) value = 'TR' + value.replace(/[^0-9]/g, '');
            else value = 'TR' + value.substring(2).replace(/[^0-9]/g, '');
            let formatted = value.match(/.{1,4}/g);
            input.value = formatted ? formatted.join(' ') : value;
        }
